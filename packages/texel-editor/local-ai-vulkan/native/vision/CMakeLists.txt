cmake_minimum_required(VERSION 3.28)
project(texel_vision_static LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(VISP_STATIC_GGML ON CACHE BOOL "" FORCE)
set(VISP_TESTS OFF CACHE BOOL "" FORCE)
set(VISP_INSTALL_MODELS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(VISP_VULKAN ON CACHE BOOL "" FORCE)
set(TEXEL_SHARED_VULKAN_RUNTIME ON)

add_subdirectory(../../third_party/vision.cpp visioncpp EXCLUDE_FROM_ALL)
target_sources(visioncpp PRIVATE ${TEXEL_MIGAN_SOURCE})
target_include_directories(visioncpp PRIVATE ${TEXEL_MIGAN_INCLUDE})
