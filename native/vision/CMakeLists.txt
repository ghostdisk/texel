cmake_minimum_required(VERSION 3.28)
project(texel_vision LANGUAGES C CXX)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(VISP_STATIC_GGML ON CACHE BOOL "" FORCE)
set(VISP_TESTS OFF CACHE BOOL "" FORCE)
set(VISP_INSTALL_MODELS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/vision.cpp visioncpp EXCLUDE_FROM_ALL)
target_sources(visioncpp PRIVATE migan.cpp)
set_target_properties(visioncpp PROPERTIES OUTPUT_NAME texel-vision)
