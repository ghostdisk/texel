cmake_minimum_required(VERSION 3.28)
project(imged_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(IMGED_VULKAN "Build the Vulkan inference backend" ON)
option(IMGED_CUDA "Build the CUDA inference backend" OFF)
set(SD_VULKAN ${IMGED_VULKAN} CACHE BOOL "" FORCE)
set(SD_CUDA ${IMGED_CUDA} CACHE BOOL "" FORCE)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(USE_TLS OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../third_party/stable-diffusion.cpp stable-diffusion)
add_subdirectory(../third_party/IXWebSocket ixwebsocket)
include(cmake/vision.cmake)
add_executable(imged-native src/main.cpp)
target_link_libraries(imged-native PRIVATE stable-diffusion ixwebsocket texel-vision)
target_compile_definitions(imged-native PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
option(IMGED_PACKAGE_RUNTIME "Install the native runtime for the Electron distribution" OFF)
if(IMGED_PACKAGE_RUNTIME)
    include(cmake/package.cmake)
endif()
